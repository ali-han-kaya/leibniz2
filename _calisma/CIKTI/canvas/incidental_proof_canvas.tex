    % ============ CONTOUR FIELD (upper 2/3) ==================
    % 24 strata bending around an invisible obstruction.
    % Base y=88, rows rise to 188, crests to ~208.9 (top margin 16).
    \begin{scope}[shift={(10,88)}]
      \def\fieldW{160} \def\obX{86}
      \foreach \i in {0,...,23}{%
        \pgfmathsetmacro{\yy}{\i*4.35}%
        \pgfmathsetmacro{\amp}{8.0*\i/23}%
        \ifnum\i=12
          \def\lineCol{verm}\else
          \pgfmathtruncatemacro{\g}{mod(\i,3)}%
          \ifcase\g\def\lineCol{fathom}\or\def\lineCol{fathom40}\or\def\lineCol{fathom25}\fi
        \fi
        \draw[\lineCol,line width=0.55pt]
          (0,\yy) .. controls (0.18*\fieldW,\yy+\amp) and (0.26*\fieldW,\yy+1.5*\amp)
                  .. (\obX-40,\yy+1.7*\amp)
               .. controls (\obX-26,\yy+1.7*\amp) and (\obX-12,\yy+0.5*\amp)
                  .. (\obX,\yy)
               .. controls (\obX+12,\yy+0.5*\amp) and (\obX+26,\yy+1.7*\amp)
                  .. (\obX+40,\yy+1.7*\amp)
               .. controls (\obX+58,\yy+1.72*\amp) and (0.92*\fieldW,\yy+0.75*\amp)
                  .. (\fieldW,\yy);
      }
      % witness: vermilion tick on S-12 + coordinates nested below the line
      \draw[verm,line width=0.9pt] (\obX,12*4.35-1.6) -- (\obX,12*4.35+1.6);
      \node[anchor=north west, text=verm, font=\ttfamily\fontsize{5.2}{6}\selectfont]
        at (\obX-10,12*4.35-0.9) {(86.0, 140.2)};
      % archive ticks, left edge of field
      \foreach \i in {0,6,12,18}{%
        \pgfmathsetmacro{\yy}{\i*4.35}%
        \draw[fathom40,line width=0.5pt] (-2.6,\yy) -- (0,\yy);}
      \node[anchor=east, text=fathom40, font=\ttfamily\fontsize{4.6}{5.6}\selectfont]
        at (-3.6,12*4.35) {S-12};
    \end{scope}

    % rotated margin annotation (left, above contour field, fully contained)
    \node[anchor=south west, text=fathom25, font=\ttfamily\fontsize{5.0}{6}\selectfont,
          rotate=90] at (6.6,162) {FIELD 01 — CONTOUR STRATA, N=24, WITNESS AT S-12};

    % ============ MEASUREMENT STRATA (lower third) ===========
    % base y=28, 10 rules, 35 columns, heights deterministic
    \begin{scope}[shift={(10,28)}]
      \def\rows{10} \def\cols{33} \def\W{160} \def\H{45}   % 34 cols x 10 bands = 340 (+1 exception = 341)
      \foreach \r in {0,...,\rows}{%
        \pgfmathsetmacro{\yy}{\r*\H/\rows}%
        \draw[fathom12,line width=0.5pt] (0,\yy) -- (\W,\yy);
      }
      \foreach \c in {0,...,\cols}{%
        \pgfmathsetmacro{\xx}{\c*\W/\cols}%
        \pgfmathsetmacro{\h}{3 + 38*mod(\c*17+11,37)/36}%
        \draw[fathom,line width=0.7pt] (\xx,0) -- (\xx,\h);
      }
      \pgfmathsetmacro{\xe}{13*\W/\cols}%
      \fill[verm] (\xe,0) rectangle ++(0.9,45);
      \node[anchor=south west, text=fathom40, font=\ttfamily\fontsize{5.0}{6}\selectfont]
        at (0,47.6) {341 MEASUREMENTS / 1 EXCEPTION};
    \end{scope}

    % ---- title block (archive plate, bottom) -----------------
    \draw[fathom,line width=1.4pt] (10,22) -- (170,22);
    \node[anchor=north west, text=fathom, font=\fontsize{22}{24}\selectfont]
      at (9.6,20.2) {INCIDENTAL PROOF};

      at (122,15.4) {PLATE 01 / SURVEY OF A BENT CLAIM};
