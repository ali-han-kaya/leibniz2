    % ============ THE RIDGE LINE (upper 2/3) ==================
    % One profile drawn three times: hairline (belief), 0.9pt
    % (doubt), vermilion (the measured season). Elevation y =
    % confidence; x = three centuries, unlabeled.
    \begin{scope}[shift={(10,84)}]
      \def\W{160} \def\H{96}
      % grid: faint vertical centuries (7 rules, unlabeled)
      \foreach \i in {1,...,7}{%
        \pgfmathsetmacro{\xx}{\i*\W/8}%
        \draw[fathom12,line width=0.4pt] (\xx,0) -- (\xx,\H);}
      % horizontal datum at the plateau of assumption
      \draw[fathom12,line width=0.5pt] (0,84) -- (\W,84);

      % --- profile 1: hairline fathom (the long belief) -------
      \draw[fathom,line width=0.35pt]
        (0,84) -- (34,84)
          .. controls (44,84) and (50,80) .. (58,79)
        -- (74,78) .. controls (86,77) and (92,64) .. (104,52)
          .. controls (112,43) and (118,42) .. (126,44)
        -- (140,47) .. controls (148,49) and (154,50) .. (\W,50);

      % --- profile 2: 0.9pt fathom40 (the doubt season) -------
      \draw[fathom40,line width=0.9pt]
        (58,79) .. controls (66,79) and (70,71) .. (74,78)
        (86,66) .. controls (92,64) and (96,56) .. (104,52);

      % --- profile 3: vermilion (the measured season) ---------
      \draw[verm,line width=1.1pt]
        (104,52) .. controls (112,43) and (118,42) .. (126,44)
        -- (140,47);

      % witness: the exact crossing where belief became measured
      \draw[verm,line width=0.9pt] (104,52-2.4) -- (104,52+2.4);
      \node[anchor=north, text=verm, font=\ttfamily\fontsize{5.2}{6}\selectfont]
        at (104,52-3.6) {(104.0, 44.0)};

      % archive ticks + section labels (left margin)
      \foreach \y/\t in {84/Y+0, 52/Y-1, 44/Y-2}{%
        \draw[fathom40,line width=0.5pt] (-2.6,\y) -- (0,\y);}
      \node[anchor=east, text=fathom40, font=\ttfamily\fontsize{4.6}{5.6}\selectfont]
        at (-3.6,84) {ASSUMED};
      \node[anchor=east, text=fathom40, font=\ttfamily\fontsize{4.6}{5.6}\selectfont]
        at (-3.6,52) {DOUBTED};
      \node[anchor=east, text=verm, font=\ttfamily\fontsize{4.6}{5.6}\selectfont]
        at (-3.6,44) {MEASURED};

      % the counterexample, drawn as weather entering from the east
      \foreach \i in {0,...,5}{%
        \pgfmathsetmacro{\wx}{150+\i*1.7}%
        \pgfmathsetmacro{\wy}{70+\i*3.4}%
        \draw[fathom25,line width=0.5pt] (\wx,\wy) -- (\wx+1.2,\wy-2.6);}
      \node[anchor=south west, text=fathom25, font=\ttfamily\fontsize{4.6}{5.6}\selectfont]

      \fill[verm] (\xv,0) rectangle ++(0.9,38);
      \node[anchor=south west, text=fathom40, font=\ttfamily\fontsize{5.0}{6}\selectfont]
        at (0,40.6) {27 SOURCES / 1 MEASURED SEASON};

    % ---- title block ------------------------------------------
    \draw[fathom,line width=1.4pt] (10,24) -- (170,24);
